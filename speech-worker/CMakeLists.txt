cmake_minimum_required(VERSION 3.21)
project(nexus-remote-speech-worker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_WITH_INSTALL_RPATH ON)
set(CMAKE_INSTALL_RPATH "$ORIGIN")
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_SDL2 OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
if(NOT WHISPER_SOURCE_DIR)
  include(FetchContent)
  FetchContent_Declare(whisper_cpp
    GIT_REPOSITORY https://github.com/ggml-org/whisper.cpp.git
    GIT_TAG 927cfce34f31707e17f2bff35c349632fb9e2c3a)
  FetchContent_MakeAvailable(whisper_cpp)
  set(WHISPER_SOURCE_DIR ${whisper_cpp_SOURCE_DIR})
else()
  add_subdirectory(${WHISPER_SOURCE_DIR} whisper-cpp)
endif()
add_executable(nexus-remote-speech-worker src/main.cpp)
target_include_directories(nexus-remote-speech-worker PRIVATE ${WHISPER_SOURCE_DIR}/examples)
target_link_libraries(nexus-remote-speech-worker PRIVATE whisper)
